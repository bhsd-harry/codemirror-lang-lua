# Operator precedence: power binds tighter than unary minus

x = -2 ^ 2

==>

Chunk(Block(AssignStatement(Name,"=",UnaryExpression(ArithOp,BinaryExpression(Number,ArithOp,Number)))))

# Power is right associative

x = 2 ^ 3 ^ 2

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(Number,ArithOp,BinaryExpression(Number,ArithOp,Number)))))

# Concatenation is right associative

x = a .. b .. c

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(Name,ConcatOp,BinaryExpression(Name,ConcatOp,Name)))))

# Multiplication binds tighter than addition

x = 1 + 2 * 3

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(Number,ArithOp,BinaryExpression(Number,ArithOp,Number)))))

# and binds tighter than or

x = a or b and c

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(Name,or,BinaryExpression(Name,and,Name)))))

# Bitwise operators, in their own order

x = a | b ~ c & d << e

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(Name,BitOp,BinaryExpression(Name,BitOp,BinaryExpression(Name,BitOp,BinaryExpression(Name,BitOp,Name)))))))

# Unary operators

x = not a
y = #t
z = ~b

==>

Chunk(Block(AssignStatement(Name,"=",UnaryExpression(not,Name)),AssignStatement(Name,"=",UnaryExpression(LengthOp,Name)),
  AssignStatement(Name,"=",UnaryExpression(BitOp,Name))))

# Comparison, including Lua's not-equal

x = a ~= b
y = a == b
z = a <= b

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(Name,CompareOp,Name)),
  AssignStatement(Name,"=",BinaryExpression(Name,CompareOp,Name)),AssignStatement(Name,"=",BinaryExpression(Name,CompareOp,Name))))

# Integer division and modulo

x = 7 // 2 % 3

==>

Chunk(Block(AssignStatement(Name,"=",BinaryExpression(BinaryExpression(Number,ArithOp,Number),ArithOp,Number))))

# Calls without parentheses

f "literal"
g {1, 2}
h [[long]]

==>

Chunk(Block(ExpressionStatement(CallExpression(Name,String)),ExpressionStatement(CallExpression(Name,TableExpression("{",Number,",",Number,"}"))),
  ExpressionStatement(CallExpression(Name,LongString))))

# Method calls chain

obj:method(1):other()

==>

("x"):upper()

==>

# Table constructor with all three field forms

t = {1, x = 2, ["y"] = 3, [4] = f(), }

==>

# Table with semicolon separators

t = {1; 2; 3}

==>

Chunk(Block(AssignStatement(Name,"=",TableExpression("{",Number,";",Number,";",Number,"}"))))

# Vararg in an expression and a table

local function f(...)
  return {...}, select("#", ...)
end

==>

Chunk(Block(LocalFunctionDeclaration(local,function,VariableDefinition,ParamList("(",Vararg,")"),Block(ReturnStatement(return,TableExpression("{",Vararg,"}"),",",CallExpression(Name,ArgList("(",String,",",Vararg,")")))),end)))

# Function expressions

local f = function(a) return a end

==>

Chunk(Block(LocalStatement(local,AttribName(VariableDefinition),"=",FunctionExpression(function,ParamList("(",VariableDefinition,")"),Block(ReturnStatement(return,Name)),end))))

# Indexing binds to the left

x = a.b[c].d
